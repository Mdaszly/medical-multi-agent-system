|-----|------|------|-----|
| 健康检查 | GET | `/api/knowledge-graph/health` | 检查图谱服务状态 |
| 症状联想 | GET | `/api/knowledge-graph/symptoms/suggest` | 输入提示 |

| 统计信息 | GET | `/api/knowledge-graph/statistics` | 获取图谱统计 |

### 功能3: 流式咨询（推荐）

**用途**: 更好的用户体验，实时显示AI响应

```javascript
async function* consultStream(sessionId, message) {
  const response = await fetch('/api/consult/stream', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({
      sessionId: sessionId || crypto.randomUUID(),
      userMessage: message
    })
  });

  const reader = response.body.getReader();
  const decoder = new TextDecoder();

  while (true) {
    const { done, value } = await reader.read();
    if (done) break;

    const chunk = decoder.decode(value);
    const lines = chunk.split('\n');

    for (const line of lines) {
      if (!line.startsWith('data: ')) continue;
      
      try {
        const data = JSON.parse(line.slice(6));
        yield data;
      } catch (e) {
        console.warn('解析SSE数据失败:', line);
      }
    }
  }
}

// React组件示例
function StreamConsult() {
  const [message, setMessage] = useState('');
  const [responses, setResponses] = useState([]);
  const [isTyping, setIsTyping] = useState(false);

  const handleSend = async () =&gt; {
    const userMessage = message;
    setMessage('');
    setIsTyping(true);
    setResponses(prev =&gt; [...prev, { type: 'user', content: userMessage }]);
    
    let aiMessage = '';
    let toolContext = null;
    
    for await (const data of consultStream(null, userMessage)) {
      switch (data.type) {
        case 'toolContext':
          toolContext = data.content;
          setResponses(prev =&gt; [...prev, { type: 'tool-context', content: toolContext }]);
          break;
        case 'delta':
          aiMessage += data.content;
          setResponses(prev =&gt; {
            const lastIdx = prev.length - 1;
            if (prev[lastIdx]?.type === 'ai-streaming') {
              const newResponses = [...prev];
              newResponses[lastIdx] = { ...newResponses[lastIdx], content: aiMessage };
              return newResponses;
            } else {
              return [...prev, { type: 'ai-streaming', content: aiMessage }];
            }
          });
          break;
        case 'done':
          setIsTyping(false);
          setResponses(prev =&gt; {
            const lastIdx = prev.length - 1;
            const newResponses = [...prev];
            newResponses[lastIdx] = { type: 'ai', content: aiMessage };
            return newResponses;
          });
          break;
      }
    }
  };

  return (
    &lt;div className="chat-container"&gt;
      &lt;div className="messages"&gt;
        {responses.map((msg, i) =&gt; (
          &lt;MessageBubble key={i} message={msg} /&gt;
        ))}
        {isTyping &amp;&amp; &lt;div className="typing-indicator"&gt;AI正在思考...&lt;/div&gt;}
      &lt;/div&gt;
      &lt;div className="input-area"&gt;
        &lt;input 
          value={message} 
          onChange={(e) =&gt; setInput(e.target.value)}
          disabled={isTyping}
        /&gt;
        &lt;button onClick={handleSend} disabled={isTyping}&gt;发送&lt;/button&gt;
      &lt;/div&gt;
    &lt;/div&gt;
  );
}

import { useState, useRef, useEffect } from 'react';

function MedicalChat() {
  const [sessionId, setSessionId] = useState(() =&gt; crypto.randomUUID());
  const [messages, setMessages] = useState([]);
  const [input, setInput] = useState('');
  const [isTyping, setIsTyping] = useState(false);
  const [suggestions, setSuggestions] = useState([]);
  const messagesEndRef = useRef(null);

  const scrollToBottom = () =&gt; {
    messagesEndRef.current?.scrollIntoView({ behavior: 'smooth' });
  };

  useEffect(scrollToBottom, [messages]);

  useEffect(() =&gt; {
    if (input.length &gt; 0) {
      const timer = setTimeout(async () =&gt; {
        const res = await fetch(
          `/api/knowledge-graph/symptoms/suggest?prefix=${encodeURIComponent(input)}`
        );
        setSuggestions(await res.json());
      }, 300);
      return () =&gt; clearTimeout(timer);
    } else {
      setSuggestions([]);
    }
  }, [input]);

  const handleSend = async () =&gt; {
    if (!input.trim() || isTyping) return;

    const userMessage = input;
    setInput('');
    setSuggestions([]);
    setIsTyping(true);
    
    setMessages(prev =&gt; [
      ...prev,
      { role: 'user', content: userMessage }
    ]);

    let aiMessage = '';
    let currentEvidence = null;
    let currentTrace = null;

    try {
      const response = await fetch('/api/consult/stream', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          sessionId,
          userMessage
        })
      });

      const reader = response.body.getReader();
      const decoder = new TextDecoder();

      while (true) {
        const { done, value } = await reader.read();
        if (done) break;

        const chunk = decoder.decode(value);
        const lines = chunk.split('\n');

        for (const line of lines) {
          if (!line.startsWith('data: ')) continue;
          
          try {
            const data = JSON.parse(line.slice(6));
            
            switch (data.type) {
              case 'toolContext':
                // 图谱证据稍后通过完整数据展示
                break;
                
              case 'delta':
                aiMessage += data.content;
                setMessages(prev =&gt; {
                  const newMessages = [...prev];
                  const lastMsg = newMessages[newMessages.length - 1];
                  if (lastMsg?.role === 'ai-streaming') {
                    lastMsg.content = aiMessage;
                    return newMessages;
                  } else {
                    return [...newMessages, { role: 'ai-streaming', content: aiMessage }];
                  }
                });
                break;
                
              case 'done':
                // 流式结束
                break;
                
              default:
                if (data.message) {
                  // 完整的ConsultVO
                  currentEvidence = data.graphEvidence;
                  currentTrace = data.agentTrace;
                  aiMessage = data.message;
                }
            }
          } catch (e) {
            console.warn('解析失败:', line);
          }
        }
      }

      // 最终消息
      setMessages(prev =&gt; {
        const newMessages = [...prev];
        if (newMessages[newMessages.length - 1]?.role === 'ai-streaming') {
          newMessages[newMessages.length - 1] = {
            role: 'ai',
            content: aiMessage,
            graphHit: currentEvidence?.graphHit,
            groundingStatus: currentEvidence ? (currentEvidence.graphHit ? 'VERIFIED' : 'NO_HIT') : null,
            evidence: currentEvidence,
            trace: currentTrace
          };
        }
        return newMessages;
      });

      padding: '20px'

        borderBottom: '1px solid #e2e8f0'

        }}&gt;
          &lt;input
            value={input}
            onChange={(e) =&gt; setInput(e.target.value)}
            onKeyDown={(e) =&gt; e.key === 'Enter' &amp;&amp; handleSend()}
            placeholder="请描述您的症状，例如：我最近总是头痛..."
            style={{
              flex: 1,
              border: 'none',
              outline: 'none',
              fontSize: '14px'
            }}
            disabled={isTyping}
          /&gt;
          &lt;button
            onClick={handleSend}
            disabled={!input.trim() || isTyping}
            style={{
              padding: '8px 20px',
              backgroundColor: '#3b82f6',
              color: 'white',
              border: 'none',
              borderRadius: '8px',
              cursor: isTyping || !input.trim() ? 'not-allowed' : 'pointer',
              opacity: isTyping || !input.trim() ? 0.5 : 1
            }}
          &gt;
            发送
          &lt;/button&gt;
        &lt;/div&gt;
      &lt;/div&gt;
    &lt;/div&gt;
  );
}

export default MedicalChat;
```
